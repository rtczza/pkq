3. 模块细化设计
3.1 网络同步与缓存管理模块设计 (NetworkCacheManager)
Rust
// src/network/cache.rs
use std::path::PathBuf;
use std::time::{Duration, SystemTime};
use ureq;

pub struct NetworkCacheManager {
    cache_dir: PathBuf, // ~/.cache/pkq/repos/
}

impl NetworkCacheManager {
    pub fn new() -> Self {
        let cache_dir = dirs::cache_dir()
            .unwrap_or_else(|| PathBuf::from("/tmp"))
            .join("pkq")
            .join("repos");
        std::fs::create_dir_all(&cache_dir).ok();
        Self { cache_dir }
    }

    /// 获取远端 URL 的元数据文件（支持 ETag 校验与 TTL）
    pub fn fetch_index(&self, repo_id: &str, url: &str, ttl: Duration, force_refresh: bool) -> Result<PathBuf, PkgError> {
        let local_path = self.cache_dir.join(repo_id).join(self.sanitize_filename(url));
        let meta_path = local_path.with_extension("meta.json");

        // 1. 检查本地缓存是否有效且未过期
        if !force_refresh && local_path.exists() {
            if let Ok(metadata) = std::fs::metadata(&local_path) {
                if let Ok(modified) = metadata.modified() {
                    if SystemTime::now().duration_since(modified).unwrap_or_default() < ttl {
                        return Ok(local_path); // 命中缓存
                    }
                }
            }
        }

        // 2. 发起原生 HTTP GET / HEAD 请求
        let mut request = ureq::get(url).timeout(Duration::from_secs(10));
        
        // 如果存在 meta 记录，带上 ETag / If-Modified-Since 请求头
        if let Ok(meta_content) = std::fs::read_to_string(&meta_path) {
            if let Ok(json) = serde_json::from_str::<HttpMeta>(&meta_content) {
                if let Some(etag) = json.etag {
                    request = request.set("If-None-Match", &etag);
                }
            }
        }

        let response = match request.call() {
            Ok(res) => res,
            Err(ureq::Error::Status(304, _)) => {
                // 304 Not Modified，更新文件修改时间并返回本地路径
                let _ = filetime::set_file_mtime(&local_path, filetime::FileTime::now());
                return Ok(local_path);
            }
            Err(e) => {
                // 网络失败降级：若本地缓存存在则使用旧缓存
                if local_path.exists() {
                    eprintln!("Warning: Network error ({}), fallback to stale cache.", e);
                    return Ok(local_path);
                }
                return Err(PkgError::NetworkError(format!("Failed to fetch {}: {}", url, e)));
            }
        };

        // 3. 写入文件并更新 HTTP 响应头元数据
        let etag = response.header("ETag").map(|s| s.to_string());
        let mut reader = response.into_reader();
        let mut out_file = std::fs::File::create(&local_path)?;
        std::io::copy(&mut reader, &mut out_file)?;

        // 保存 ETag 元数据
        let http_meta = HttpMeta { etag };
        let _ = std::fs::write(meta_path, serde_json::to_string(&http_meta)?);

        Ok(local_path)
    }

    fn sanitize_filename(&self, url: &str) -> String {
        // 将 URL 转为安全的缓存文件名
        url.replace("://", "_").replace('/', "_")
    }
}

#[derive(Serialize, Deserialize)]
struct HttpMeta {
    etag: Option<String>,
}
3.2 RPM 适配器实时网络工作流
解析 /etc/yum.repos.d/*.repo：获取远程 baseurl (例如 [https://mirrors.kernel.org/fedora/releases/39/Everything/x86_64/os/](https://mirrors.kernel.org/fedora/releases/39/Everything/x86_64/os/))。

请求 repomd.xml：调用 NetworkCacheManager.fetch_index(repo_id, "https://.../repodata/repomd.xml", ttl, force)。

定位并请求 primary.xml.gz：从 repomd.xml 解析出最新的 primary.xml.gz 相对路径，发起网络 HTTP 下载并缓存。

解析元数据：使用 flate2 + quick-xml 读取缓存的 .xml.gz 文件，组装为 PkgMetadata。

3.3 DEB 适配器实时网络工作流
解析 /etc/apt/sources.list：提取 Base URL、发行版代号（如 jammy）和组件（main）。

请求 Packages.gz：调用网络模块拼装 URL（例如 [http://archive.ubuntu.com/ubuntu/dists/jammy/main/binary-amd64/Packages.gz](http://archive.ubuntu.com/ubuntu/dists/jammy/main/binary-amd64/Packages.gz)）发起 HTTP 请求下载。

解析文本：利用流式解压读取 Packages control 文本并加载元数据。

4. 关键业务流程流程图文字描述

4.1 流程：实时在线仓库搜索完整流程 (search --repo)
用户执行 pkq search nginx --repo --refresh。

CLI 接收到 --refresh，构造 CacheConfig { force_refresh: true, ... }。

Backend 遍历系统配置的 Repo 列表，将每个 Repo 的 URL 提交给 NetworkCacheManager。

NetworkCacheManager 使用 ureq HTTP 客户端并发/顺序连接远程镜像源：

检查 HTTP ETag，若未改变（304）直接复用本地缓存。

若索引已更新（200 OK），流式写入下载数据至 ~/.cache/pkq/repos/。

适配器解析最新的 XML / Control 文本，匹配包含 nginx 的远端软件包。

输出线上最新仓库中的真实包元数据列表。

4.2 流程：版本对比 (diff)

用户执行 `pkq diff nginx --refresh`。

CLI 构造 CacheConfig { force_refresh: true, ... }，调用 Backend.diff_versions("nginx", cfg)。

Backend 分别从本地数据源（rpmdb / dpkg status）和线上仓库（primary.xml / Packages.gz）获取 nginx 的版本信息。

对比 VersionDiff { local_version, repo_version, is_outdated } 并格式化输出。

5.3 RPM 本地数据源设计

| 项目 | 说明 |
|------|------|
| 数据库路径 | `/var/lib/rpm/rpmdb.sqlite`（SQLite，RHEL 8+）或 `/var/lib/rpm/Packages`（BDB，RHEL 7-） |
| 解析方式 | SQLite：通过 `rusqlite` 只读查询；BDB：降级为仅读仓库索引（`--repo`） |
| 文件列表 | 从 rpmdb `Provide`/`Basenames` 表查询，等价于 `rpm -ql` |
| 变更日志 | 从 rpmdb `Changelog` 表查询 |
